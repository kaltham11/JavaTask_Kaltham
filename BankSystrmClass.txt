import java.io.FileNotFoundException;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;
import java.io.File;
import java.io.IOException;
import java.io.FileWriter;

public class BankSystem {
    private List<Account> accounts; //list to save the account

    public BankSystem() {
        accounts = new ArrayList<>();//constructor
    }

    public void createAccount() {
        Scanner scanner = new Scanner(System.in);
        System.out.println("Please, Enter your Civil ID");
        String civilId = scanner.nextLine();
        for (Account checkAcc : accounts) {
            if (checkAcc.getCivilId().equals(civilId))
                System.out.println("you already have an account, your Account is : " + checkAcc.getAccountNumber());
            return;//to stop the methods
        }
        System.out.println("Please, Enter your Name");
        String name = scanner.nextLine();


        System.out.println("Please, Enter your Number Phone");
        String numberPhone = scanner.nextLine();

        String accountNumber = generateUniqeAccountNumber();
        Double balance = 0.0;

        LocalDate openDate = LocalDate.now();

        String branch = "";
        int branchChoice = -1;
        while (true) {
            System.out.println("Please, select the branch number");
            System.out.println("1. Muscat Branch");
            System.out.println("2. Salalah Branch");
            System.out.println("3. Sohar Branch");
            System.out.println("4. Nizwa Branch");
            System.out.println("5. Sur Branch");

            try {
                branchChoice = Integer.parseInt(scanner.nextLine());
            } catch (NumberFormatException e) {
                System.out.println("Please! Enter a valid number");
                continue;
            }

            switch (branchChoice) {
                case 1:
                    branch = "Muscat Branch";
                    break;

                case 2:
                    branch = "Salalah Branch";
                    break;

                case 3:
                    branch = "Sohar Branch";
                    break;

                case 4:
                    branch = "Nizwa Branch";
                    break;

                case 5:
                    branch = "Sur Branch";
                    break;
                default:
                    System.out.println("Invalied choice");
                    continue;
            }
            break;
        }

        Account account = new Account(name, civilId, numberPhone, accountNumber, balance, openDate, branch);

        accounts.add(account);
        System.out.println("your Account create successfully");
        System.out.println("your Account number is: " + accountNumber);

    }
    private static long accountNum = 12000453678531000L;
    private String generateUniqeAccountNumber() {
        accountNum++;
        return String.valueOf(accountNum);

    }

    public void DepositMoney() {
        Scanner scanner = new Scanner(System.in);
        System.out.println("Please, Enter your civil Id");
        String civilId = scanner.nextLine();
        for (Account checkAcc : accounts) {
            if (checkAcc.getCivilId().equals(civilId)) {
                System.out.println("Please Enter the money");
                Double amount = scanner.nextDouble();
                if (amount > 0) {
                    checkAcc.setBalance(checkAcc.getBalance() + amount);
                    checkAcc.addTransaction("Deposited:" + amount + " on " + java.time.LocalDate.now());
                    System.out.println(amount + " has been debited from your account " + checkAcc.getAccountNumber() + " Your available balance is=\t" + checkAcc.getBalance());
                } else {

                    System.out.println("The amount must be grater than zero");
                }

            }
            return;
        }
        System.out.println("Account not found with civil ID:" + civilId);
    }

    public void withdrawMoney() {
        Scanner scanner = new Scanner(System.in);
        System.out.println("Please, Enter your civil Id");
        String civilId = scanner.nextLine();
        for (Account checkAcc : accounts) {
            if (checkAcc.getCivilId().equals(civilId)) {
                System.out.println("Please Enter the money");
                Double amount = scanner.nextDouble();
                if (amount <= 0) {
                    System.out.println("The amount must be grater than zero");
                    return;
                }
                if (amount > checkAcc.getBalance()) {
                    System.out.println(amount + "Insufficient process, because your available balance is =\t" + checkAcc.getBalance());

                } else {
                    checkAcc.setBalance(checkAcc.getBalance() - amount);
                    checkAcc.addTransaction("Withdrew:" + amount + " on " + java.time.LocalDate.now());
                    System.out.println(amount + " has been withdrawn from " + checkAcc.getAccountNumber() + " " +
                            " Your available balance is=\t" + checkAcc.getBalance());
                }

                return;
            }

        }
        System.out.println("Account not found with civil ID:" + civilId);
    }

    public void viewTransactionHistory() {
        Boolean found = false;
        Scanner scanner = new Scanner(System.in);
        System.out.println("Please, Enter your civil ID");
        String civilId = scanner.nextLine();
        for (Account checkAcc : accounts) {
            if (checkAcc.getCivilId().equals(civilId)) {
                System.out.println("Transction History for account " + checkAcc.getAccountNumber());
                List<String> history = checkAcc.getTransactionHistory();
                if (history.isEmpty()) {
                    System.out.println("No Transaction History found");
                } else {
                    for (String record : history) {
                        System.out.println("- " + record);
                    }
                }
                return;
            }
        }

        System.out.println("Account not found with civil ID:" + civilId);
    }

    public void saveData() {
        try {
            File myObj = new File("SaveData.txt");
            if (myObj.createNewFile()) {
                System.out.println("File created: " + myObj.getName());
            } else {
                System.out.println("File already exists.");
            }
        } catch (IOException e) {
            System.out.println("An error occurred.");
            e.printStackTrace();
        }
        try {
            FileWriter myWriter = new FileWriter("SaveData.txt");
            for (Account checkAcc : accounts) {
                String line = "Name: " + checkAcc.getName() + "\n"
                        + "Account Number: " + checkAcc.getAccountNumber() + "\n"
                        + "Transaction: " + checkAcc.getTransactionHistory() + "\n";
                myWriter.write(line + "\n");
            }

            myWriter.close();
            System.out.println("Successfully wrote to the file.");
        } catch (IOException e) {
            System.out.println("An error occurred.");
            e.printStackTrace();
        }
    }

    public void loadData() {
        try {
            File myObj = new File("SaveData.txt");
            Scanner scanner = new Scanner(myObj);
            while (true) {

                if (!scanner.hasNextLine()) break;
                String nameLine = scanner.nextLine();

                if (!scanner.hasNextLine()) break;
                String accountLine = scanner.nextLine();

                if (!scanner.hasNextLine()) break;
                String transactionLine = scanner.nextLine();

                String name = nameLine.substring(nameLine.indexOf(":") + 2);
                String accountNumber = accountLine.substring(accountLine.indexOf(":") + 2);

                Account account = new Account(name, accountNumber);

                int start = transactionLine.indexOf("[");
                int end = transactionLine.indexOf("]");
                if (start != -1 && end != -1 && end > start) {
                    String transaction = transactionLine.substring(start + 1, end);

                    String[] transactions = transaction.split(", ");

                    for (String tran:transactions){
                        account.addTransaction(tran.trim());
                    }
                }
                accounts.add(account);
            }
            scanner.close();
            System.out.println("Data loaded successfully");
        } catch (FileNotFoundException e) {
            System.out.println("File not found");
            e.printStackTrace();
        } catch (Exception e) {
            System.out.println("Error while loading the file");
            e.printStackTrace();
        }

    }
}
